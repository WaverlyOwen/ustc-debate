\begin{stage}{2. 正一被质询防守预案}{反四质询，120 秒双边计时}
\textbf{原则}：<回答短、守住前提、不否认常识>
\begin{qa}
\qarow{<对方会问>}{<我方回答（简短，守住前提）>}
\end{qa}
\end{stage}

\begin{stage}{3. 正四质询反一问题链}{120 秒，双边计时}
\textbf{总目标}：<拿到什么承认 / 暴露什么矛盾>
\begin{chain}{链一：<名称>}{<目标>}
\q{<问题，25 字内>}{答“是”：<下一问>；答“否”：<下一问>}
\cut{<打断用语>}
\close{<收束语>}
\end{chain}
\end{stage}

\begin{kit}{4. 正二驳论}{套件 · 组装后 NNN–NNN 字 / 120 秒}
\assembly{开头 → 回应反二（任选 1） → 驳对方立论（任选 2） → 收尾}
\begin{fixed}{开头}<一句话点出我方要打的层面，不预设对方说了什么>\end{fixed}
% 只有后驳论的一方（正二）有这一组
\begin{pick}{回应反二}{1}
\begin{module}{如果反二打的是 <我方哪一点>}<回应>\end{module}
\begin{fallback}{反二没有打到我方论点}<一句带过，把时间留给驳立论>\end{fallback}
\end{pick}
\begin{pick}{驳对方立论}{2}
\begin{module}{如果对方立论建立在 <主干一> 上}<对方说了什么 → 问题在哪 → 在我方判准下应当怎么看>\end{module}
\begin{module}{如果对方立论建立在 <主干二> 上}<…>\end{module}
\begin{fallback}{对方立论的主干不在以上几条时}<打对方的判准：不管对方立论用什么论据都成立的那一击>\end{fallback}
\end{pick}
\begin{fixed}{收尾}<回到我方判准与一句话立论>\end{fixed}
\end{kit}

\begin{points}{备用点（6–8 个）}
\item <…>
\end{points}
\begin{points}{对方最可能问的三个问题 → 回应}
\item <…>
\end{points}
\end{stage}

\begin{stage}{6. 正三盘问问题链}{90 秒，单边计时}
\begin{chain}{链一：<名称>}{问二辩 → 问四辩同一问题，检验口径}
\q{（问反二）<…>}{<…>}
\q{（问反四）同样的问题}{<两人若不一致，小结重锤>}
\end{chain}
\textbf{盘问目标}：<小结里要证明什么>
\end{stage}

\begin{stage}{7. 正二、正四被盘问防守预案}{两人口径必须一致}
\begin{qaa}
\qaarow{<对方会问>}{<二辩答>}{<四辩答（必须一致）>}
\end{qaa}
\end{stage}

\begin{kit}{8. 正三盘问小结}{套件 · 组装后 NNN–NNN 字 / 90 秒}
\assembly{链一结果（任选 1） → 链二结果（任选 1） → 接回我方论点}
% 后小结的一方（反三）在最前面多一个「回应正三小结」选择组
\begin{pick}{链一的结果}{1}
\begin{module}{如果对方承认了 <…>}<这说明什么>\end{module}
\begin{module}{如果对方否认了 <…>}<这说明什么>\end{module}
\begin{module}{如果对方回避了}<回避本身就是成果：点出这个问题对方没有正面回答，请评委记住。问了几次、原话是什么，条件保证不了，不写>\end{module}
\end{pick}
\begin{pick}{链二的结果}{1}
\begin{module}{如果对方承认了 <…>}<…>\end{module}
\begin{fallback}{链二没有问完}<一句带过，转入接回论点>\end{fallback}
\end{pick}
\begin{fixed}{接回我方论点}<盘问指向我方哪个论点、哪条判准>\end{fixed}
\end{kit}

\begin{stage}{9. 自由辩战场设计}{240 秒}
\begin{battleground}{战场一：<名称>（前 90 秒，主战场）}
\begin{fields}
\field{目标}{<…>}
\field{开场问题}{<封闭式，20 字内>}
\field{对方答 A → 追问}{<…>}
\field{对方答 B → 追问}{<…>}
\field{对方可能反攻 → 我方防守}{<…>}
\field{收束语}{<…>}
\end{fields}
\end{battleground}
\begin{points}{必问（前两分钟内问完）}
\item <…>
\end{points}
\begin{points}{必答（15 秒以内正面回答）}
\item <问题 → 答案>
\end{points}
\textbf{时间分配与站位}：<…>
\end{stage}

\begin{kit}{10. 正四结辩}{套件 · 组装后 NNN–NNN 字 / 210 秒}
\assembly{归纳分歧（任选 1） → 处理最强点（任选 1） → 回应反四（任选 1） → 临场 → 论点收束 → 价值升华 → 结尾}
\begin{pick}{归纳分歧}{1}
\begin{module}{如果全场主战场落在 <…>}<双方真正的争点是什么>\end{module}
\begin{fallback}{主战场不在以上几条时}<用我方判准重新框定分歧>\end{fallback}
\end{pick}
\begin{pick}{处理对方最强点}{1}
\begin{module}{如果对方全场最有力的是 <…>}<正面回应，内容取自反驳库>\end{module}
\begin{fallback}{对方最有力的一点不在以上几条时}<重申我方最硬的一条论据>\end{fallback}
\end{pick}
% 正四写「回应反四」；反四把这一组换成「封路」：条件是「如果正方全场主打……」，
% 内容是这条路的问题与请评委使用的检验标准，不写对方待会儿会说什么。
\begin{pick}{回应反四}{1}
\begin{module}{如果反四收在 <…>}<从反四实际说的一句话切入，回应>\end{module}
\begin{fallback}{反四的收束不在以上几条时}<重申判准，交给临场位>\end{fallback}
\end{pick}
\live{60}   % 只有正四留：回应反四结辩里无法预判的内容。紧跟回应组，不插进价值升华与结尾之间
\begin{fixed}{论点收束}<两个论点在交锋后为什么仍然成立。不引用台上具体交锋>\end{fixed}
\begin{fixed}{价值升华}<定稿。具体的人和时刻，不是宏大叙事。全场最值得打磨的一段>\end{fixed}
\begin{fixed}{结尾}<一句能被记住的短句>\end{fixed}
\end{kit}
\begin{contingency}
\ifthen{<时间不够>}{<先删哪个部件；价值升华不删>}
\end{contingency}
